using System;
using System.Text;
using System.Web.UI;
using System.Web.UI.WebControls;

namespace OnlineEventRegistrationForm
{
    public partial class RegistrationForm : Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
            ValidationSettings.UnobtrusiveValidationMode =
                UnobtrusiveValidationMode.None;
        }

        protected void btnSubmit_Click(object sender, EventArgs e)
        {
            if (!Page.IsValid)
                return;

            if (!chkTerms.Checked)
            {
                lblMessage.Text =
                    "Please accept Terms and Conditions.";

                lblMessage.ForeColor =
                    System.Drawing.Color.Red;

                pnlResult.Visible = true;
                return;
            }

            StringBuilder output = new StringBuilder();

            output.Append("<b>Full Name:</b> " + txtName.Text + "<br/>");
            output.Append("<b>Email:</b> " + txtEmail.Text + "<br/>");
            output.Append("<b>Contact No:</b> " + txtContact.Text + "<br/>");
            output.Append("<b>Department:</b> " +
                rblDepartment.SelectedItem.Text + "<br/>");
            output.Append("<b>Event:</b> " +
                ddlEvent.SelectedItem.Text + "<br/>");
            output.Append("<b>Gender:</b> " +
                rblGender.SelectedItem.Text + "<br/>");

            string skills = "";

            foreach (ListItem item in cblSkills.Items)
            {
                if (item.Selected)
                {
                    if (skills != "")
                        skills += ", ";

                    skills += item.Text;
                }
            }

            output.Append("<b>Skills:</b> " + skills + "<br/>");
            output.Append("<b>Address:</b> " +
                txtAddress.Text.Replace("\r\n", "<br/>"));

            lblMessage.Text = "Registration Successful!";
            lblMessage.ForeColor =
                System.Drawing.Color.Green;

            lblOutput.Text = output.ToString();

            pnlResult.Visible = true;
        }

        protected void cvSkills_ServerValidate(
            object source,
            ServerValidateEventArgs args)
        {
            bool selected = false;

            foreach (ListItem item in cblSkills.Items)
            {
                if (item.Selected)
                {
                    selected = true;
                    break;
                }
            }

            args.IsValid = selected;
        }

        protected void cvTerms_ServerValidate(
            object source,
            ServerValidateEventArgs args)
        {
            args.IsValid = chkTerms.Checked;
        }
    }
}
