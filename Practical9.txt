2. Default.aspx.cs
Location: Default.aspx.cs
using System;
using System.Configuration;
using System.Data;
using System.Data.SqlClient;

namespace PlacementReport
{
    public partial class Default : System.Web.UI.Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                LoadStudents();
            }
        }

        private void LoadStudents()
        {
            string cs = ConfigurationManager
                .ConnectionStrings["PlacementDBConnectionString"]
                .ConnectionString;

            string query = @"
                SELECT StudentID, StudentName,
                Department, Company, Package, Status
                FROM Students
                ORDER BY StudentID";

            using (SqlConnection con = new SqlConnection(cs))
            {
                using (SqlDataAdapter da =
                    new SqlDataAdapter(query, con))
                {
                    DataTable dt = new DataTable();

                    da.Fill(dt);

                    GridView1.DataSource = dt;
                    GridView1.DataBind();
                }
            }
        }
    }
}

4. Companys.aspx.cs
Location: Companys.aspx.cs
using System;
using System.Configuration;
using System.Data;
using System.Data.SqlClient;

namespace PlacementReport
{
    public partial class Companys : System.Web.UI.Page
    {
        protected void Page_Load(object sender, EventArgs e)
        {
            if (!IsPostBack)
            {
                LoadPlacements();
            }
        }

        private void LoadPlacements()
        {
            string cs = ConfigurationManager
                .ConnectionStrings["PlacementDBConnectionString"]
                .ConnectionString;

            using (SqlConnection con = new SqlConnection(cs))
            {
                // Company placement details
                string query = @"
                    SELECT Company, Package, StudentID,
                    StudentName, Department
                    FROM Students
                    WHERE Status = 'Placed'
                    ORDER BY Company";

                using (SqlDataAdapter da =
                    new SqlDataAdapter(query, con))
                {
                    DataTable dt = new DataTable();

                    da.Fill(dt);

                    GridView1.DataSource = dt;
                    GridView1.DataBind();
                }

                // Company details summary
                string summaryQuery = @"
                    SELECT Company, Package,
                    COUNT(StudentID) AS TotalStudents
                    FROM Students
                    WHERE Status = 'Placed'
                    GROUP BY Company, Package
                    ORDER BY Company";

                using (SqlDataAdapter da2 =
                    new SqlDataAdapter(summaryQuery, con))
                {
                    DataTable dt2 = new DataTable();

                    da2.Fill(dt2);

                    GridView2.DataSource = dt2;
                    GridView2.DataBind();
                }
            }
        }
    }
}

5. Web.config
Location: Web.config

Add this connection string inside the existing configuration tag.
<connectionStrings>
    <add name="PlacementDBConnectionString"
         connectionString="Data Source=(localdb)\MSSQLLocalDB;Initial Catalog=PlacementDB;Integrated Security=True"
         providerName="System.Data.SqlClient" />
</connectionStrings>

6. SQL Server Database
Open SQL Server Object Explorer and execute:
CREATE DATABASE PlacementDB;
GO

USE PlacementDB;
GO

CREATE TABLE Students
(
    StudentID INT PRIMARY KEY,
    StudentName VARCHAR(100),
    Department VARCHAR(50),
    Company VARCHAR(100),
    Package DECIMAL(10,2),
    Status VARCHAR(20)
);
